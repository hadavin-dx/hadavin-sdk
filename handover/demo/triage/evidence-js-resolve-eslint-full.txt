function load(f, root) {
  const out = [];
  for (const r of JSON.parse(fs.readFileSync(f, 'utf8'))) {
    const rel = r.filePath.slice(root.length + 1), lines = fs.readFileSync(r.filePath, 'utf8').split('\n');

  return out;
}
