def get(u, t=3):
    try:
        with urllib.request.urlopen(u, timeout=t) as r: return r.status, r.read().decode()
    except urllib.error.HTTPError as e: return e.code, e.read().decode()
    except Exception as e: return type(e).__name__, ""
def post(env, t=15):
    r = urllib.request.Request(D + "/leads", json.dumps(env).encode(), {"Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(r, timeout=t) as x: return x.status, x.read().decode()[:300]
    except urllib.error.HTTPError as e: return e.code, e.read().decode()[:300]
    except Exception as e: return type(e).__name__, str(getattr(e, "reason", ""))[:80]
def up(url, tries=40):
    for _ in range(tries):
        if get(url)[0] == 200: return True
        time.sleep(0.5)
    return False
procs = []
def start(args, log):
    p = subprocess.Popen([PY] + args, cwd=ROOT, env=ENV, stdout=open(log, "ab"), stderr=subprocess.STDOUT, start_new_session=True); procs.append(p); return p
def stop_all():
    for p in procs:
        try: os.killpg(p.pid, signal.SIGTERM)
        except Exception: pass
    time.sleep(1.5)
    for p in procs:
        try: os.killpg(p.pid, signal.SIGKILL)
        except Exception: pass
    procs.clear()
def clean():
    for f in glob.glob(ROOT + "/data/*"):

def db_counts():
    out = {}
    for f in [x for x in glob.glob(ROOT + "/data/*") if os.path.isfile(x) and open(x, "rb").read(15) == b"SQLite format 3"]:
        c = sqlite3.connect("file:%s?mode=ro" % f, uri=True)
        for t in ("receipts", "leads", "followups"):
            try: out[t] = c.execute("select count(*) from %s" % t).fetchone()[0]
            except Exception as e: out[t] = "err"
        try: out["followup_rows"] = [tuple(r) for r in c.execute("select * from followups").fetchall()][:3]
        except Exception: pass
        try: out["followup_cols"] = [d[0] for d in c.execute("select * from followups limit 1").description]
        except Exception: pass
        c.close()
    return out
