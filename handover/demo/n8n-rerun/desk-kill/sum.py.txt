import json,glob,re,collections
def cnt(s):
    m=re.search(r'"count": (\d+)',s); return int(m.group(1)) if m else '?'
def fu(s):
    m=re.search(r'"(pending|failed|unknown|sending)_followups": (\d+)',s); 
    return {k:int(v) for k,v in re.findall(r'"(pending_followups|failed_followups|unknown_followups|sending_followups)": (\d+)',s)}
rows=[]
for f in sorted(glob.glob('/tmp/rr/dk/result-*.json')):
    for k in json.load(open(f)):
        a=k['after']; e=k['email']
        # per-email counts: crm/outbox bodies truncated; use count (one lead per fresh book)
        rows.append((k['point'],k['kind'],k['i'],k['desk_exit_code'],k['first_call'][0],k['second_call'][0],cnt(a['crm']),cnt(a['outbox']),a['db'].get('leads'),a['db'].get('receipts'),{x:y for x,y in fu(a['counts']).items() if y}))
for r in rows: print(r)
print()
g=collections.Counter((r[0],r[1],r[6],r[7],r[8],json.dumps(r[10])) for r in rows)
