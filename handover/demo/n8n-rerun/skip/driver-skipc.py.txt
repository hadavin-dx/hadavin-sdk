def unflat(s):
    arr = json.loads(s); cache = {}
    def res(i):
        if i in cache: return cache[i]
        v = arr[i]
        if isinstance(v, list):
            out = cache[i] = []; out.extend(res(int(x)) if isinstance(x, str) else x for x in v); return out
        if isinstance(v, dict):
            out = cache[i] = {}; out.update({k: (res(int(x)) if isinstance(x, str) else x) for k, x in v.items()}); return out
        return v
    return res(0)
def post(path, lid, timeout=60):
    r = urllib.request.Request(f"<loopback>/webhook/{path}", json.dumps({"lead_id": lid}).encode(), {"Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(r, timeout=timeout) as x: return x.status, x.read().decode()[:400]
    except urllib.error.HTTPError as e: return e.code, e.read().decode()[:400]
    except Exception as e: return type(e).__name__, ""
def burst(path, lid, n=10):
    bar = threading.Barrier(n); out = []
    def go():
        bar.wait(); out.append(post(path, lid))
    ts = [threading.Thread(target=go) for _ in range(n)]; [t.start() for t in ts]; [t.join() for t in ts]; return out
def exec_view(e_id):
    d = api.call("GET", f"/executions/{e_id}")
    d = d.get("data", d)
    if "HTTPError" in d: return {"id": e_id, "err": d}
    runs = (unflat(d["data"]).get("resultData", {}).get("runData", {})) if isinstance(d.get("data"), str) else {}
    nodes = {}
    for nm, rl in runs.items():
        r0 = rl[0]; main = (r0.get("data") or {}).get("main") or []
        nodes[nm] = {"status": r0.get("executionStatus"), "outputs": [None if o is None else len(o) for o in main], "first": (main[0][0]["json"] if main and main[0] else None)}
    return {"id": d.get("id", e_id), "status": d.get("status"), "nodes": nodes}
def max_id(wid): return max([int(x["id"]) for x in lib.execs(wid, None, 100)] or [0])
def execs_since(wid, base):
    time.sleep(1.5)
    for _ in range(20):
        rows = [x for x in lib.execs(wid, None, 100) if int(x["id"]) > base]
        if rows and not any(x.get("status") in ("running", "new", "waiting") for x in rows): break
        time.sleep(1)
    return [exec_view(x["id"]) for x in sorted(rows, key=lambda x: x.get("startedAt") or "")]
def iso(): return time.strftime("%Y-%m-%dT%H:%M:%S", time.gmtime(time.time() - 2))
def count(path, lid):

    return psql("SELECT count(*) FROM leads_skip WHERE lead_id='%s'" % lid)
