The migration scanner refuses that (`migrate.py:87-90`, `_UNREAD` contains `AS IDENTITY`; `_scan` 123-147), so the column is `BIGSERIAL`: the scanner then registers the sequence `kill_switch_stop_requests_id_seq` (read: `migrate.py:145-146`), and it needs its own `GRANTS` entry `Grant("sequence", "")`. - NEW view `kill_switch_state` (ONE source for "is the switch on"; created by the static migration; the runtime, the stopper and the reset group get `SELECT` on the VIEW only; `Grant.kind` accepts "view", read `migrate.py:23-37`):
  ```sql
  CREATE VIEW kill_switch_state AS
  SELECT (k."on" OR r.id IS NOT NULL) AS "on",
         CASE WHEN k."on" THEN k."by" WHEN r.id IS NOT NULL THEN r."by" ELSE k."by" END AS "by",
         CASE WHEN k."on" THEN k.ts   WHEN r.id IS NOT NULL THEN r.ts   ELSE k.ts   END AS ts,
         k.schema_version, k.version,
         (SELECT count(*) FROM kill_switch_stop_requests WHERE cleared_by IS NULL)::int AS open_requests
  FROM kill_switch k
  LEFT JOIN LATERAL (SELECT id, "by", ts FROM kill_switch_stop_requests
                     WHERE cleared_by IS NULL ORDER BY id LIMIT 1) r ON true
  WHERE k.id = 1;
  ```
  The switch is ON when `kill_switch."on"` is true OR at least one request has `cleared_by IS NULL`. `by` and `ts` come from the flag when it is on, else from the EARLIEST open request. `version` and `open_requests` are the pair the operator passes back to reset.   CREATE OR REPLACE FUNCTION <S>.hadavin_kill_switch_reset(p_by text, p_expected_version bigint, p_expected_open int)
  RETURNS TABLE ("on" boolean, "by" text, ts timestamptz, schema_version int, version bigint, open_requests int)

      RAISE EXCEPTION 'kill switch: the version changed' USING ERRCODE = 'HD003'; END IF;
    UPDATE <S>.kill_switch_stop_requests r SET cleared_by = v + 1 WHERE r.cleared_by IS NULL;   -- ONE statement
    GET DIAGNOSTICS n = ROW_COUNT;

  END $f$;

  CREATE OR REPLACE FUNCTION <S>.hadavin_kill_switch_guard() RETURNS trigger
  LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, pg_temp AS $f$
  DECLARE lbl text;
  BEGIN
    IF OLD."on" AND NOT NEW."on" THEN
      IF NOT pg_has_role(session_user, '<GROUP>'::text, 'MEMBER') THEN
        RAISE EXCEPTION 'kill switch: only the reset role may clear it' USING ERRCODE = 'HD001'; END IF;
      SELECT label INTO lbl FROM <S>.kill_switch_login_labels WHERE login_name = session_user::text;
      IF lbl IS NULL THEN RAISE EXCEPTION 'kill switch: login not registered' USING ERRCODE = 'HD001'; END IF;
      INSERT INTO <S>.kill_switch_resets VALUES (OLD.version, now(), lbl, OLD."by");
      NEW."by" := lbl; NEW.version := OLD.version + 1; NEW.ts := now();
    ELSIF NEW."on" AND NOT OLD."on" THEN
      IF NEW."by" IS NOT DISTINCT FROM OLD."by" THEN NEW."by" := NULL; END IF;   -- "by" not set by this update: drop the last resetter's label (round 8 M1)
      NEW.version := OLD.version; NEW.ts := now();      -- no bump: an old release's stop (a raw upsert) still works after a rollback
    ELSE
      NEW."on" := OLD."on"; NEW."by" := OLD."by"; NEW.ts := OLD.ts; NEW.version := OLD.version;
    END IF;
    NEW.id := OLD.id; RETURN NEW;
  END $f$;
