def rec(probe, where, out, markers):

    rows.append((probe, where, "LEAK" if leaked else "no marker", out[:160].replace("\n", " | ")))

    f.write_text(f.read_text(encoding="utf-8").replace("model_id: fake-model", 'model_id: "%s"' % val), encoding="utf-8")

    with contextlib.redirect_stdout(o), contextlib.redirect_stderr(e):
        try:
            rc = cli.main(["--config", str(d), "check-config"])
        except BaseException as exc:
            rc = "EXC"; e.write(type(exc).__name__ + ": " + str(exc) + repr(exc) + "".join(traceback.format_exception(exc)))
    rec("check-config model_id=" + shape.replace("{m}", "<M>"), "rc=%s stdout+stderr" % rc, o.getvalue() + e.getvalue(), [mk])
