def classify(text: str) -> str:
    """Mechanical label only (first rule that holds); the source line is printed beside it for a reader."""
    t = text.strip()
    if t in TABLE_KIND_NAMES or t in SOURCE_KIND_NAMES:
        return "kind name from the table"
    if "://" in t or t.startswith("/"):
        return "URL path"
    if re.fullmatch(r"[A-Z][A-Z0-9_]*", t):
        return "environment-variable name (ALL_CAPS token)"
    if re.search(r"\s", t):
        return "prose (contains whitespace)"
    if re.fullmatch(r"[A-Za-z0-9_.:\-{}\[\]]+", t):
        return "config key (single lower/mixed token, not a table kind)"
    return "other (no rule matched)"


def cell(s: str, n: int = 110) -> str:
    s = repr(s)
    if len(s) > n:

CAMEL = re.compile(r"[A-Z]+(?![a-z])\d*|[A-Z]?[a-z]+\d*|\d+")


def parts(name: str) -> list[str]:
    """Snake parts (split on `_`), the camel parts of each snake part, and each pair of neighbouring camel
    parts joined (so GitHub -> github, GitLab -> gitlab, OpenAI/OpenAi -> openai)."""
    out = []
    for snake in [s for s in name.split("_") if s]:
        out.append(snake.lower())
        camel = [c.lower() for c in CAMEL.findall(snake)]
        out += camel
        out += [a + b for a, b in zip(camel, camel[1:], strict=False)]
    return out


def targets(node) -> list[ast.Name]:
    if isinstance(node, ast.Assign):
        tg = node.targets
    elif isinstance(node, ast.AnnAssign | ast.AugAssign):
        tg = [node.target]
    else:
        return []
    names = []
    for t in tg:
        names += [x for x in ast.walk(t) if isinstance(x, ast.Name)]
    return names

def run(label, fn):
    print(f"### {label}")
    try:
        print(f"returned: {fn()!r}")
    except Exception:  # recorded raw, not judged
        print("raised:")
        print("```")
        print(traceback.format_exc().rstrip())
        print("```")
    print()
